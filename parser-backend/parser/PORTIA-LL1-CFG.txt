================================================================================
SECTION 1: PROGRAM STRUCTURE
================================================================================
// STRICT SECTION ORDERING (enforced syntactically):
//   1. Global declarations (0+) - must start with GLOBAL keyword
//   2. Bare top-level arrays (0+) - type id[size] = {...};
//   3. Weave definitions (0+) - weave type definitions
//   4. Weave instances (0+) - instances of weave types
//   5. Function declarations (0+) - all func definitions
//   6. Main block (exactly 1, required, must be last)
//
// IMPORTANT RESTRICTIONS:
//   - Bare scalar declarations FORBIDDEN at top level (e.g., "int x = 5;")
//   - Bare arrays ALLOWED: int arr[3] = {1,2,3}; (bracket required after id)
//   - STRICT TYPE-LITERAL MATCHING: declarations accept ONLY matching literals
//   - Expressions FORBIDDEN in all declaration contexts
//   - EMPTY INITIALIZERS FORBIDDEN: {} is syntactically impossible

<program> -> <global_section>

--------------------------------------------------------------------------------
1.1 Global Section
--------------------------------------------------------------------------------
// Processes globals/weaves/arrays, then transitions to functions or main
// TYPE ID at top level requires [ (array) - scalar declarations forbidden

<global_section> -> <global_decl> <global_section>
<global_section> -> int id <int_array_with_init> ; <global_section>
<global_section> -> long id <long_array_with_init> ; <global_section>
<global_section> -> float id <float_array_with_init> ; <global_section>
<global_section> -> double id <double_array_with_init> ; <global_section>
<global_section> -> char id <char_array_with_init> ; <global_section>
<global_section> -> string id <string_array_with_init> ; <global_section>
<global_section> -> bool id <bool_array_with_init> ; <global_section>
<global_section> -> weave id { <field_list> } ; <global_section>
<global_section> -> id <weave_inst_decl> <global_section>
<global_section> -> <function_decl> <func_and_main>
<global_section> -> int main ( ) { <main_body> }

--------------------------------------------------------------------------------
1.2 Function Section
--------------------------------------------------------------------------------
// One or more functions, then mandatory main
// Once here, no more globals allowed - strict section ordering enforced

<func_and_main> -> <function_decl> <func_and_main>
<func_and_main> -> int main ( ) { <main_body> }

--------------------------------------------------------------------------------
1.3 Global Declarations
--------------------------------------------------------------------------------
// MUST start with GLOBAL keyword
// Each type accepts ONLY its matching literal type (no expressions)

<global_decl> -> global <mutability> int id = intlit <int_global_cont> ;
<global_decl> -> global <mutability> long id = longlit <long_global_cont> ;
<global_decl> -> global <mutability> float id = floatlit <float_global_cont> ;
<global_decl> -> global <mutability> double id = doublelit <double_global_cont> ;
<global_decl> -> global <mutability> char id = charlit <char_global_cont> ;
<global_decl> -> global <mutability> string id = stringlit <string_global_cont> ;
<global_decl> -> global <mutability> bool id = <bool_lit> <bool_global_cont> ;

// Multi-variable continuation (type-specific literal enforcement)
<int_global_cont> -> , id = intlit <int_global_cont>
<int_global_cont> -> lambda

<long_global_cont> -> , id = longlit <long_global_cont>
<long_global_cont> -> lambda

<float_global_cont> -> , id = floatlit <float_global_cont>
<float_global_cont> -> lambda

<double_global_cont> -> , id = doublelit <double_global_cont>
<double_global_cont> -> lambda

<char_global_cont> -> , id = charlit <char_global_cont>
<char_global_cont> -> lambda

<string_global_cont> -> , id = stringlit <string_global_cont>
<string_global_cont> -> lambda

<bool_global_cont> -> , id = <bool_lit> <bool_global_cont>
<bool_global_cont> -> lambda

--------------------------------------------------------------------------------
1.4 Mutability and Boolean Literal
--------------------------------------------------------------------------------

<mutability> -> var
<mutability> -> const

<bool_lit> -> true
<bool_lit> -> false

SECTION 5: CONTROL STRUCTURES
================================================================================
// NON-EMPTY BODY ENFORCEMENT: All control structures require >=1 statement
// switch uses <arg_expr> for condition expression

--------------------------------------------------------------------------------
5.1 Control Structures for Main Function
--------------------------------------------------------------------------------

<ctrl_struct> -> if ( <condition> ) { <non_empty_ctrl_stmt_list> } <else_opt>
<ctrl_struct> -> switch ( <arg_expr> ) { <case_list> <default_opt> }
<ctrl_struct> -> for ( <for_init> ; <for_cond> ; <for_update> ) { <non_empty_loop_ctrl_stmt_list> }
<ctrl_struct> -> while ( <condition> ) { <non_empty_loop_ctrl_stmt_list> }
<ctrl_struct> -> do { <non_empty_loop_ctrl_stmt_list> } while ( <condition> ) ;

<ctrl_stmt_list> -> <statement_non_return> <ctrl_stmt_list> | lambda
<non_empty_ctrl_stmt_list> -> <statement_non_return> <ctrl_stmt_list>
<loop_ctrl_stmt_list> -> <loop_statement_non_return> <loop_ctrl_stmt_list> | lambda
<non_empty_loop_ctrl_stmt_list> -> <loop_statement_non_return> <loop_ctrl_stmt_list>

<else_opt> -> else <else_body> | lambda
<else_body> -> { <non_empty_ctrl_stmt_list> }
<else_body> -> if ( <condition> ) { <non_empty_ctrl_stmt_list> } <else_opt>

<case_list> -> case <case_val> : <non_empty_loop_ctrl_stmt_list> <break_opt> <case_list> | lambda
<case_val> -> intlit | longlit | charlit | true | false
<default_opt> -> default : <non_empty_loop_ctrl_stmt_list> <break_opt> | lambda
<break_opt> -> break ; | lambda

--------------------------------------------------------------------------------
5.2 Type-Specific Control Structures
--------------------------------------------------------------------------------
// Each return type has dedicated control structure rules
// Pattern repeated for: int, long, float, double, char, string, bool, array, weave, void
// Example shown for INT only - others follow identical pattern

<ctrl_struct_int> -> if ( <condition> ) { <non_empty_stmt_list_int> } <else_opt_int>
<ctrl_struct_int> -> switch ( <arg_expr> ) { <case_list_int> <default_opt_int> }
<ctrl_struct_int> -> for ( <for_init> ; <for_cond> ; <for_update> ) { <non_empty_loop_stmt_list_int> }
<ctrl_struct_int> -> while ( <condition> ) { <non_empty_loop_stmt_list_int> }
<ctrl_struct_int> -> do { <non_empty_loop_stmt_list_int> } while ( <condition> ) ;

<stmt_list_int> -> <statement_int> <stmt_list_int> | lambda
<non_empty_stmt_list_int> -> <statement_int> <stmt_list_int>
<loop_statement_int> -> <statement_int> | break ;
<loop_stmt_list_int> -> <loop_statement_int> <loop_stmt_list_int> | lambda
<non_empty_loop_stmt_list_int> -> <loop_statement_int> <loop_stmt_list_int>

<else_opt_int> -> else <else_body_int> | lambda
<else_body_int> -> { <non_empty_stmt_list_int> }
<else_body_int> -> if ( <condition> ) { <non_empty_stmt_list_int> } <else_opt_int>

<case_list_int> -> case <case_val> : <non_empty_loop_stmt_list_int> <break_opt> <case_list_int> | lambda
<default_opt_int> -> default : <non_empty_loop_stmt_list_int> <break_opt> | lambda

SECTION 9: EXPRESSIONS

--------------------------------------------------------------------------------
9.1 Typed Expression Dispatch System
--------------------------------------------------------------------------------
// Expression routing enforces type discipline at the syntax level:
//   - Assignment does NOT chain (a = b = c is syntax error)
//   - Boolean operators (&&, ||) require boolean operands
//   - Concat (..) requires string-like operands
//   - Relational (<, >, <=, >=) require numeric operands
//   - Arithmetic (+, -, *, /, %) require numeric operands

<expression> -> <typed_assign_expr>

<typed_assign_expr> -> <typed_concat_expr> <typed_assign_tail>
// Simple assignment (=) allows any RHS; compound ops restricted to numeric
<typed_assign_tail> -> = <typed_rhs_expr>
<typed_assign_tail> -> += <typed_numeric_add_expr>
<typed_assign_tail> -> -= <typed_numeric_add_expr>
<typed_assign_tail> -> *= <typed_numeric_add_expr>
<typed_assign_tail> -> /= <typed_numeric_add_expr>
<typed_assign_tail> -> %= <typed_numeric_add_expr>
<typed_assign_tail> -> lambda

<assign_op> -> = | += | -= | *= | /= | %=

<typed_rhs_expr> -> <typed_concat_expr>

--------------------------------------------------------------------------------
9.2 Typed Dispatch (based on starting token)
--------------------------------------------------------------------------------

<typed_concat_expr> -> stringlit <typed_string_cont>
<typed_concat_expr> -> charlit <typed_string_cont>
<typed_concat_expr> -> intlit <typed_numeric_cont>
<typed_concat_expr> -> longlit <typed_numeric_cont>
<typed_concat_expr> -> floatlit <typed_numeric_cont>
<typed_concat_expr> -> doublelit <typed_numeric_cont>
<typed_concat_expr> -> true <typed_bool_cont>
<typed_concat_expr> -> false <typed_bool_cont>
<typed_concat_expr> -> ! <typed_bool_factor> <typed_bool_tail_opt>
<typed_concat_expr> -> - <typed_neg_numeric_cont>
<typed_concat_expr> -> id <typed_id_cont>
<typed_concat_expr> -> ( <typed_paren_cont>
<typed_concat_expr> -> int ( <expression> ) <typed_numeric_cont>
<typed_concat_expr> -> long ( <expression> ) <typed_numeric_cont>
<typed_concat_expr> -> float ( <expression> ) <typed_numeric_cont>
<typed_concat_expr> -> double ( <expression> ) <typed_numeric_cont>
<typed_concat_expr> -> char ( <expression> ) <typed_string_cont>
<typed_concat_expr> -> string ( <expression> ) <typed_string_cont>
<typed_concat_expr> -> bool ( <expression> ) <typed_bool_cont>
<typed_concat_expr> -> ++ id
<typed_concat_expr> -> -- id

//  typed_id_cont, typed_paren_cont, typed_postfix_chain, etc.]

9.3 Typed Return Expressions
--------------------------------------------------------------------------------

// NUMERIC RETURN (int, long, float, double)
<typed_numeric_ret_expr> -> <typed_numeric_add_expr>

// STRING RETURN (char, string)  
<typed_string_ret_expr> -> <typed_string_ret_primary> <typed_string_cont>

// BOOL RETURN
<typed_bool_ret_expr> -> <typed_bool_ret_primary> <typed_bool_ret_tail>

--------------------------------------------------------------------------------
9.4 Numeric Expression Rules
--------------------------------------------------------------------------------

<typed_numeric_add_expr> -> <typed_numeric_mul_expr> <typed_numeric_add_tail>
<typed_numeric_add_tail> -> + <typed_numeric_mul_expr> <typed_numeric_add_tail> | - ... | lambda

<typed_numeric_mul_expr> -> <typed_numeric_unary_expr> <typed_numeric_mul_tail>
<typed_numeric_mul_tail> -> * <typed_numeric_unary_expr> <typed_numeric_mul_tail> | / ... | % ... | lambda

<typed_numeric_unary_expr> -> ! <typed_numeric_unary_expr>
<typed_numeric_unary_expr> -> - <typed_numeric_unary_expr>
<typed_numeric_unary_expr> -> <typed_numeric_postfix_expr>

--------------------------------------------------------------------------------
9.5 Postfix Chain
--------------------------------------------------------------------------------

<typed_postfix_chain> -> [ <array_index> ] <typed_postfix_after_arr>
<typed_postfix_chain> -> . id <typed_postfix_chain>
<typed_postfix_chain> -> ( <arg_list> ) <typed_postfix_chain>
<typed_postfix_chain> -> lambda

<typed_postfix_after_arr> -> [ <array_index> ] | . id <typed_postfix_chain> | ( <arg_list> ) <typed_postfix_chain> | lambda

================================================================================
SECTION 10: EFFECTFUL STATEMENTS
================================================================================
// Only allows constructs with side effects: assignment, ++/--, function calls
// Rejects: bare id; arr[i]; obj.field; literals; pure expressions

<effect_stmt> -> ++ id <effect_pre_chain>
<effect_stmt> -> -- id <effect_pre_chain>
<effect_stmt> -> id <effect_id_cont>

<effect_pre_chain> -> [ <stmt_array_index> ] <effect_pre_arr_chain>
<effect_pre_chain> -> . id <effect_pre_chain>
<effect_pre_chain> -> lambda

// Simple assignment (=) allows any RHS; compound ops restricted to numeric
<effect_id_cont> -> = <stmt_assign_expr>
<effect_id_cont> -> += <numeric_add_expr_stmt>
<effect_id_cont> -> -= <numeric_add_expr_stmt>
<effect_id_cont> -> *= <numeric_add_expr_stmt>
<effect_id_cont> -> /= <numeric_add_expr_stmt>
<effect_id_cont> -> %= <numeric_add_expr_stmt>
<effect_id_cont> -> ++
<effect_id_cont> -> --
<effect_id_cont> -> ( <stmt_arg_list> ) <effect_post_call>
<effect_id_cont> -> [ <stmt_array_index> ] <effect_post_arr>
<effect_id_cont> -> . id <effect_post_member>

// Array element assignment - compound ops restricted to numeric
<effect_arr_effect> -> = <stmt_assign_expr>
<effect_arr_effect> -> += <numeric_add_expr_stmt>
<effect_arr_effect> -> -= <numeric_add_expr_stmt>
<effect_arr_effect> -> *= <numeric_add_expr_stmt>
<effect_arr_effect> -> /= <numeric_add_expr_stmt>
<effect_arr_effect> -> %= <numeric_add_expr_stmt>
<effect_arr_effect> -> ++ | -- | ( <stmt_arg_list> ) <effect_post_call> | . id <effect_post_member>

// Field assignment - compound ops restricted to numeric
<effect_post_member> -> = <stmt_assign_expr>
<effect_post_member> -> += <numeric_add_expr_stmt>
<effect_post_member> -> -= <numeric_add_expr_stmt>
<effect_post_member> -> *= <numeric_add_expr_stmt>
<effect_post_member> -> /= <numeric_add_expr_stmt>
<effect_post_member> -> %= <numeric_add_expr_stmt>
<effect_post_member> -> ++ | -- | ( <stmt_arg_list> ) <effect_post_call>
<effect_post_member> -> [ <stmt_array_index> ] <effect_post_arr> | . id <effect_post_member>

<arg_expr> -> <arg_typed_rhs> <arg_assign_tail>
<arg_assign_tail> -> <assign_op> <arg_typed_rhs> | lambda

<arg_typed_rhs> -> <arg_bool_or_concat>

11.1 Argument Layer Numeric Rules
--------------------------------------------------------------------------------

<numeric_mul_tail_arg> -> * <numeric_unary_expr_arg> <numeric_mul_tail_arg> | / ... | % ... | lambda

<numeric_add_expr_arg> -> <numeric_mul_expr_arg> <numeric_add_tail_arg>
<numeric_add_tail_arg> -> + <numeric_mul_expr_arg> <numeric_add_tail_arg> | - ... | lambda

<numeric_unary_expr_arg> -> ! <numeric_unary_expr_arg>
<numeric_unary_expr_arg> -> - <numeric_unary_expr_arg>
<numeric_unary_expr_arg> -> <numeric_postfix_expr_arg>

<numeric_postfix_expr_arg> -> ( <arg_expr> ) <arg_postfix_chain>

<arg_array_access_dim2> -> [ <arg_array_index> ] | lambda
<arg_array_index> -> intlit | id

<arg_nested_tail> -> , <arg_expr> <arg_nested_tail> | lambda

<numeric_add_expr_stmt> -> <numeric_mul_expr_stmt> <numeric_add_tail_stmt>
<numeric_add_tail_stmt> -> + <numeric_mul_expr_stmt> <numeric_add_tail_stmt> | - ... | lambda

<numeric_unary_expr_stmt> -> ! <numeric_unary_expr_stmt>
<numeric_unary_expr_stmt> -> - <numeric_unary_expr_stmt>

<stmt_postfix_chain> -> ( <stmt_arg_list> ) <stmt_postfix_chain>
<stmt_postfix_chain> -> lambda

================================================================================
SECTION 13: CONDITION EXPRESSIONS (STRICT BOOLEAN DISCIPLINE)
================================================================================
// Conditions enforce STRICT BOOLEAN semantics:
//   - ONLY boolean-producing constructs allowed at top level
//   - Relational expressions ALWAYS require comparison operator
//   - Arithmetic expressions FORBIDDEN without comparison
//   - Type casts FORBIDDEN without comparison
//   - Numeric literals FORBIDDEN standalone
//
// ALLOWED:
//   if (x)              - Boolean identifier
//   if (true)           - Boolean literal
//   if (x < 5)          - Relational expression
//   if (x && y)         - Logical combination
//   if (!x)             - Negation
//   if ((x + 5) > 10)   - Paren arithmetic with comparison
//
// DISALLOWED (syntax error):
//   if (2)              - Standalone numeric literal
//   if (x + 5)          - Standalone arithmetic
//   if (int(5))         - Standalone type cast

<condition> -> <cond_or>

<cond_or> -> <cond_and> <cond_or_tail>
<cond_or_tail> -> || <cond_and> <cond_or_tail> | lambda

<cond_and> -> <cond_not> <cond_and_tail>
<cond_and_tail> -> && <cond_not> <cond_and_tail> | lambda

<cond_not> -> ! <cond_not>
<cond_not> -> <cond_atom>

<cond_atom> -> true | false
<cond_atom> -> id <cond_id_cont>
<cond_atom> -> ( <cond_paren_inner> ) <cond_paren_tail>
<cond_atom> -> <cond_lit_cmp>

<cond_id_cont> -> [ <cond_arr_index> ] <cond_id_arr_cont>
<cond_id_cont> -> + <cond_rhs_unary> <cond_rhs_mul> <cond_rhs_add> <cond_cmp> <cond_rhs>
<cond_id_cont> -> - <cond_rhs_unary> <cond_rhs_mul> <cond_rhs_add> <cond_cmp> <cond_rhs>
<cond_id_cont> -> * <cond_rhs_unary> <cond_rhs_mul> <cond_rhs_add> <cond_cmp> <cond_rhs>
<cond_id_cont> -> / <cond_rhs_unary> <cond_rhs_mul> <cond_rhs_add> <cond_cmp> <cond_rhs>
<cond_id_cont> -> % <cond_rhs_unary> <cond_rhs_mul> <cond_rhs_add> <cond_cmp> <cond_rhs>
<cond_id_cont> -> < <cond_rhs> | > <cond_rhs> | <= <cond_rhs> | >= <cond_rhs> | == <cond_rhs> | != <cond_rhs>
<cond_id_cont> -> lambda

// Literal-starting comparison (MUST have comparison - no bare literals)
<cond_lit_cmp> -> intlit <cond_lit_mul> <cond_lit_add> <cond_cmp> <cond_rhs>
<cond_lit_cmp> -> longlit <cond_lit_mul> <cond_lit_add> <cond_cmp> <cond_rhs>
<cond_lit_cmp> -> floatlit <cond_lit_mul> <cond_lit_add> <cond_cmp> <cond_rhs>
<cond_lit_cmp> -> doublelit <cond_lit_mul> <cond_lit_add> <cond_cmp> <cond_rhs>
<cond_lit_cmp> -> - <cond_lit_unary> <cond_lit_mul> <cond_lit_add> <cond_cmp> <cond_rhs>

<cond_cmp> -> < | > | <= | >= | == | !=
