PORTIA LL(1)-Compliant Context-Free Grammar
============================================

Terminals: global, local, func, return, if, else, switch, case, default, for, while, do, break,
           trap, thread, threadln, using, weave, main, int, long, float, double, char, string,
           bool, void, var, const, true, false, id, intlit, longlit, floatlit, doublelit,
           charlit, stringlit, =, +=, -=, *=, /=, %=, ==, !=, <, >, <=, >=, &&, ||, !,
           +, -, *, /, %, ++, --, .., (, ), {, }, [, ], ;, ,, :, .

============================================
PROGRAM STRUCTURE
============================================
// Left-factored to resolve INT lookahead conflict between int_decl and main_func
// main_func terminates program (nothing allowed after)

<program> → <decl_list>

// When INT is seen, look ahead to distinguish ID (variable) vs MAIN (main function)
<decl_list> → int <int_decl_or_main>
<decl_list> → <other_decl> <decl_list>

// After INT: ID continues with typed_decl_tail then more decls, MAIN ends program
<int_decl_or_main> → id <typed_decl_tail> <decl_list>
<int_decl_or_main> → main ( ) { <main_body> } $END

// All non-INT declarations (no ambiguity)
<other_decl> → global <mutability> <dtype> id = <expression> <multi_global> ;
<other_decl> → long id <typed_decl_tail>
<other_decl> → float id <typed_decl_tail>
<other_decl> → double id <typed_decl_tail>
<other_decl> → char id <typed_decl_tail>
<other_decl> → string id <typed_decl_tail>
<other_decl> → bool id <typed_decl_tail>
<other_decl> → weave id { <field_list> } ;
<other_decl> → id <weave_inst_decl>
<other_decl> → func <nonvoid_ret_type> id ( <param_list> ) { <function_body_nonvoid> }
<other_decl> → func void id ( <param_list> ) { <function_body_void> }

// main_func rule for reference (matched via int_decl_or_main)
<main_func> → int main ( ) { <main_body> }

<multi_global> → , id = <expression> <multi_global>
<multi_global> → λ

<weave_inst_decl> → id <weave_inst_tail> <weave_inst_cont> ;
<weave_inst_decl> → <array_with_init> <weave_arr_cont> ;

<weave_inst_tail> → = { <weave_field_value> <weave_field_list_tail> }
<weave_inst_tail> → <array_with_init>

<weave_field_value> → intlit
<weave_field_value> → longlit
<weave_field_value> → floatlit
<weave_field_value> → doublelit
<weave_field_value> → charlit
<weave_field_value> → stringlit
<weave_field_value> → true
<weave_field_value> → false
<weave_field_value> → { <weave_arr_content> }

<weave_arr_content> → { <elem_list> } <elem_2d_tail>
<weave_arr_content> → <arr_elem> <elem_1d_tail>
<weave_arr_content> → λ

<weave_field_list_tail> → , <weave_field_value> <weave_field_list_tail>
<weave_field_list_tail> → λ

<weave_inst_cont> → , id <weave_inst_tail> <weave_inst_cont>
<weave_inst_cont> → λ

<weave_arr_cont> → , id <array_with_init> <weave_arr_cont>
<weave_arr_cont> → λ

<typed_decl_tail> → <array_with_init> ;
<typed_decl_tail> → = <expression> <multi_typed> ;

<multi_typed> → , id = <expression> <multi_typed>
<multi_typed> → λ

UNIFIED EXPRESSION GRAMMAR
============================================

<assign_op> → =

<assign_op> → -=
<assign_op> → *=
<assign_op> → /=
<assign_op> → %=

<concat_expr> → <or_expr> <concat_tail>

<concat_tail> → .. <or_expr> <concat_tail>
<concat_tail> → λ

============================================

<or_expr> → <and_expr> <or_tail>

<break_opt> → λ

<for_init_type> → int

<for_init_type> → float
<for_init_type> → double
<for_init_type> → char

<for_update_tail> → --

============================================
CONDITION EXPRESSIONS (STRICT BOOLEAN)
============================================
// Conditions are strictly boolean - NOT arithmetic expressions
// Valid conditions: comparisons, logical ops (&&, ||, !), bool literals, function calls
// Invalid: standalone arithmetic (x, x+1, x*2) - must use comparison operator
// Assignment operators (= += -= *= /= %=) are PROHIBITED in conditions

<condition> → <cond_or>

<cond_or> → <cond_and> <cond_or_tail>
<cond_or_tail> → || <cond_and> <cond_or_tail>
<cond_or_tail> → λ

<cond_and> → <cond_not> <cond_and_tail>
<cond_and_tail> → && <cond_not> <cond_and_tail>
<cond_and_tail> → λ

<cond_not> → ! <cond_not>
<cond_not> → <cond_atom>

// Condition RHS expression: starts at concat_expr, bypassing assign_expr
// This prohibits = += -= *= /= %= on the RHS of comparisons within conditions
<cond_rhs_expr> → <concat_expr>

// Condition atom: must be comparison, bool literal, bool function call, or parenthesized condition
// Standalone arithmetic expressions are NOT valid
<cond_atom> → true
<cond_atom> → false
<cond_atom> → ( <condition> )
<cond_atom> → id <cond_after_id>
<cond_atom> → intlit <mul_tail> <add_tail> <comp_op> <cond_rhs_expr>
<cond_atom> → longlit <mul_tail> <add_tail> <comp_op> <cond_rhs_expr>
<cond_atom> → floatlit <mul_tail> <add_tail> <comp_op> <cond_rhs_expr>
<cond_atom> → doublelit <mul_tail> <add_tail> <comp_op> <cond_rhs_expr>
<cond_atom> → charlit <mul_tail> <add_tail> <comp_op> <cond_rhs_expr>
<cond_atom> → stringlit <mul_tail> <add_tail> <comp_op> <cond_rhs_expr>
<cond_atom> → int ( <expression> ) <mul_tail> <add_tail> <comp_op> <cond_rhs_expr>
<cond_atom> → long ( <expression> ) <mul_tail> <add_tail> <comp_op> <cond_rhs_expr>
<cond_atom> → float ( <expression> ) <mul_tail> <add_tail> <comp_op> <cond_rhs_expr>
<cond_atom> → double ( <expression> ) <mul_tail> <add_tail> <comp_op> <cond_rhs_expr>
<cond_atom> → char ( <expression> ) <mul_tail> <add_tail> <comp_op> <cond_rhs_expr>
<cond_atom> → string ( <expression> ) <mul_tail> <add_tail> <comp_op> <cond_rhs_expr>
<cond_atom> → bool ( <expression> ) <mul_tail> <add_tail> <comp_op> <cond_rhs_expr>

// After id in condition: function call (standalone bool or comparison) OR identifier for comparison
<cond_after_id> → ( <arg_list> ) <cond_after_call>
<cond_after_id> → <cond_after_id_no_call>

// After id(): standalone bool function OR continue to comparison
<cond_after_call> → λ
<cond_after_call> → <postfix_chain> <mul_tail> <add_tail> <comp_op> <cond_rhs_expr>

// After id not followed by ( : build expression with context-aware comparison requirement
<cond_after_id_no_call> → <cond_postfix_no_call> <cond_arith_final>

// Arithmetic-aware comparison: standalone bool OR comparison OR arithmetic+mandatory comparison
<cond_arith_final> → λ
<cond_arith_final> → <comp_op> <cond_rhs_expr>
<cond_arith_final> → <mul_arith> <add_tail> <comp_op> <cond_rhs_expr>
<cond_arith_final> → <add_arith> <comp_op> <cond_rhs_expr>

// Non-empty arithmetic operations (require comparison)
<mul_arith> → * <unary_expr> <mul_tail>
<mul_arith> → / <unary_expr> <mul_tail>
<mul_arith> → % <unary_expr> <mul_tail>

<add_arith> → + <mul_expr> <add_tail>
<add_arith> → - <mul_expr> <add_tail>

// Postfix operations after id (excluding direct call since that's handled separately)
<cond_postfix_no_call> → [ <array_index> ] <array_access_dim2> <postfix_after_arr>
<cond_postfix_no_call> → . id <postfix_chain>
<cond_postfix_no_call> → ++
<cond_postfix_no_call> → --
<cond_postfix_no_call> → λ

<comp_op> → ==
<comp_op> → !=
<comp_op> → <
<comp_op> → >
<comp_op> → <=
<comp_op> → >=

<main_content> → return intlit ;
